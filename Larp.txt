-- Smooth Mobile Dragging
local dragging, dragInput, dragStart, startPos
Topbar.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = Main.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    end
end)

Topbar.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
        dragInput = input
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if input == dragInput and dragging then
        local delta = input.Position - dragStart
        Main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
end)

-- Smooth Toggle Animation
local open = true
local isAnimating = false

local function ToggleUI()
    if isAnimating then return end
    isAnimating = true
    
    if open then
        local tweenSize = TweenService:Create(Main, TweenInfo.new(0.25, Enum.EasingStyle.Quart, Enum.EasingDirection.In), {Size = UDim2.new(0, 210, 0, 0)})
        tweenSize:Play()
        tweenSize.Completed:Connect(function()
            Main.Visible = false
            open = false
            isAnimating = false
        end)
    else
        Main.Visible = true
        local tweenSize = TweenService:Create(Main, TweenInfo.new(0.3, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), {Size = UDim2.new(0, 210, 0, 135)})
        tweenSize:Play()
        tweenSize.Completed:Connect(function()
            open = true
            isAnimating = false
        end)
    end
end

-- 1. ESP System
local ESP_Enabled = false

local function RemoveESP(player)
    if player.Character then
        local head = player.Character:FindFirstChild("Head")
        if head and head:FindFirstChild("LarpESP_Billboard") then
            head.LarpESP_Billboard:Destroy()
        end
        if player.Character:FindFirstChild("LarpESP_Highlight") then
            player.Character.LarpESP_Highlight:Destroy()
        end
    end
end

local function ApplyESP(player)
    if not ESP_Enabled or player == LocalPlayer then return end

    local function CharacterAdded(char)
        if not char or not ESP_Enabled then return end
        
        local humanoid = char:WaitForChild("Humanoid", 5)
        local head = char:WaitForChild("Head", 5)

        if not humanoid or not head then return end

        if not head:FindFirstChild("LarpESP_Billboard") then
            local billboard = Instance.new("BillboardGui")
            billboard.Name = "LarpESP_Billboard"
            billboard.Adornee = head
            billboard.Size = UDim2.new(0, 150, 0, 40)
            billboard.StudsOffset = Vector3.new(0, 2.5, 0)
            billboard.AlwaysOnTop = true
            billboard.Parent = head

            local nameLabel = Instance.new("TextLabel", billboard)
            nameLabel.Size = UDim2.new(1, 0, 0, 18)
            nameLabel.Position = UDim2.new(0, 0, 0, 0)
            nameLabel.BackgroundTransparency = 1
            nameLabel.Text = player.Name
            nameLabel.TextColor3 = CyanColor
            nameLabel.TextSize = 12
            nameLabel.Font = Enum.Font.GothamBold
            nameLabel.TextStrokeTransparency = 0.2

            local hpLabel = Instance.new("TextLabel", billboard)
            hpLabel.Name = "HPLabel"
            hpLabel.Size = UDim2.new(1, 0, 0, 16)
            hpLabel.Position = UDim2.new(0, 0, 0, 18)
            hpLabel.BackgroundTransparency = 1
            hpLabel.Text = "HP: " .. math.floor(humanoid.Health) .. " / " .. math.floor(humanoid.MaxHealth)
            hpLabel.TextColor3 = CyanColor
            hpLabel.TextSize = 10
            hpLabel.Font = Enum.Font.GothamMedium
            hpLabel.TextStrokeTransparency = 0.2

            humanoid.HealthChanged:Connect(function(health)
                hpLabel.Text = "HP: " .. math.floor(health) .. " / " .. math.floor(humanoid.MaxHealth)
            end)
        end

        if not char:FindFirstChild("LarpESP_Highlight") then
            local highlight = Instance.new("Highlight")
            highlight.Name = "LarpESP_Highlight"
            highlight.Adornee = char
            highlight.FillColor = CyanColor
            highlight.FillTransparency = 0.5
            highlight.OutlineColor = CyanColor
            highlight.OutlineTransparency = 0
            highlight.Parent = char
        end
    end

    if player.Character then
        task.spawn(function() CharacterAdded(player.Character) end)
    end
    player.CharacterAdded:Connect(CharacterAdded)
end

-- 2. Legit Auto QTE Logic
local LegitAutoQTEEnabled = false
local LegitAutoQTELoop = nil

local function toggle_legit_auto_qte(enabled)
    LegitAutoQTEEnabled = enabled
    if LegitAutoQTELoop then
        task.cancel(LegitAutoQTELoop)
        LegitAutoQTELoop = nil
    end
    if enabled then
        LegitAutoQTELoop = task.spawn(function()
            local QTE = nil
            local success, module = pcall(function()
                return require(ReplicatedStorage:WaitForChild("Modules"):WaitForChild("HBGQTE"))
            end)
            if success then
                QTE = module
            else
                LegitAutoQTEEnabled = false
                TweenService:Create(QTESwitchBG, TweenInfo.new(0.2), {BackgroundColor3 = Color3.fromRGB(35, 35, 35)}):Play()
                TweenService:Create(QTESwitchCircle, TweenInfo.new(0.2), {Position = UDim2.new(0, 2, 0.5, -5), BackgroundColor3 = Color3.fromRGB(150, 150, 150)}):Play()
                return
            end
            while LegitAutoQTEEnabled do
                task.wait(0.05)
                pcall(function()
                    if QTE and QTE.ActiveButtons then
                        for _, data in pairs(QTE.ActiveButtons) do
                            if data and data.Inner and data.Outer
                                and not data.Inner:GetAttribute("Tweening")
                                and not data.Inner:GetAttribute("Failed") then
                                local innerSize = data.Inner.AbsoluteSize.X
                                local outerSize = data.Outer.AbsoluteSize.X
                                if innerSize > 0 and outerSize > 0
                                    and math.abs(innerSize - outerSize) <= 12 then
                                    pcall(function() QTE.Pressed(false, data) end)
                                end
                            end
                        end

-- 6. Advanced Auto Escape Logic
EscapeBtn.MouseButton1Click:Connect(function()
    local char = LocalPlayer.Character
    if not char then return end
    
    local root = char:FindFirstChild("HumanoidRootPart")
    if not root then return end

    local targetCFrame = nil

    for _, v in pairs(workspace:GetDescendants()) do
        if v:IsA("SpawnLocation") and v.Enabled then
            targetCFrame = v.CFrame + Vector3.new(0, 3, 0)
            break
        end
    end

    if not targetCFrame then
        for _, v in pairs(workspace:GetDescendants()) do
            if v:IsA("BasePart") then
                local name = v.Name:lower()
                if name:find("spawn") or name:find("base") or name:find("lobby") or name:find("safezone") then
                    targetCFrame = v.CFrame + Vector3.new(0, 4, 0)
                    break
                end
            end
        end
    end

    if targetCFrame then
        root.CFrame = targetCFrame
    else
        root.CFrame = CFrame.new(0, 100, 0)
    end
end)
